import { MetadataRoute } from 'next'

export default function robots(): MetadataRoute.Robots {
  return {
    rules: {
      userAgent: '*',
      allow: '/',
      disallow: ['/api/', '/admin/'], // Don't crawl private backend folders
    },
    sitemap: 'https://jntuhresultbp.vercel.app/sitemap.xml',
  }
}
